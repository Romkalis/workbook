Binary search tree: дерево поиска

У каждого узла есть data, left и right. Правило BST: всё, что меньше узла, лежит слева, всё, что больше, — справа. Узлы создаются классом Node из extensions/list-tree.js: раскомментируйте его импорт в заготовке. Здесь ловушки с прототипом нет: тесты дерева сравнивают node.data, а не узлы целиком.

root() и add()

Обратите внимание: root — метод, а не свойство, тест вызывает tree.root().data. Поэтому сам корень храним в приватном поле #root, иначе имя поля и метода совпадёт.

const { Node } = require('../extensions/list-tree.js');

class BinarySearchTree {
  #root = null;

  root() {
    return this.#root;
  }

  add(data) {
    const node = new Node(data);
    if (!this.#root) {
      this.#root = node;
      return;
    }
    let current = this.#root;
    while (true) {
      if (data === current.data) return;
      const side = data < current.data ? 'left' : 'right';
      if (!current[side]) {
        current[side] = node;
        return;
      }
      current = current[side];
    }
  }

Спускаемся от корня: меньше — налево, больше — направо, пока не найдём пустое место. Повтор значения просто игнорируем. Задание не говорит, что делать с дубликатами, а для has и find так проще всего.

find(), has(), min(), max()

  find(data) {
    let current = this.#root;
    while (current) {
      if (data === current.data) return current;
      current = data < current.data ? current.left : current.right;
    }
    return null;
  }

  has(data) {
    return this.find(data) !== null;
  }

  min() {
    if (!this.#root) return null;
    let current = this.#root;
    while (current.left) current = current.left;
    return current.data;
  }

  max() {
    if (!this.#root) return null;
    let current = this.#root;
    while (current.right) current = current.right;
    return current.data;
  }

find возвращает сам узел, а не значение: тест проверяет tree.find(8).data. has не дублирует обход, а переиспользует find. Минимум — самый левый узел, максимум — самый правый. Для пустого дерева оба возвращают null.

remove(): три случая

Удаление — главная сложность задания. Сначала находим узел, затем смотрим, сколько у него детей:

  1. Нет детей (лист). Просто убираем: родитель получает null.
  2. Один ребёнок. Ставим ребёнка на место удалённого узла.
  3. Два ребёнка. Находим преемника — минимальный узел правого поддерева. Он больше всего слева и меньше всего справа, поэтому может встать на место удаляемого без нарушения правила. Копируем его значение в узел, а самого преемника удаляем из правого поддерева. У преемника нет левого ребёнка, так что это случай 1 или 2.
  remove(data) {
    this.#root = removeNode(this.#root, data);
  }
}

function removeNode(node, data) {
  if (!node) return null;
  if (data < node.data) {
    node.left = removeNode(node.left, data);
    return node;
  }
  if (data > node.data) {
    node.right = removeNode(node.right, data);
    return node;
  }
  if (!node.left) return node.right;
  if (!node.right) return node.left;
  let successor = node.right;
  while (successor.left) successor = successor.left;
  node.data = successor.data;
  node.right = removeNode(node.right, successor.data);
  return node;
}

module.exports = { BinarySearchTree };

Рекурсивная функция возвращает новый корень поддерева, и родитель записывает его себе: node.left = removeNode(...). Поэтому не нужно отдельно хранить родителя, и удаление корня работает так же, как любого другого узла: this.#root = removeNode(this.#root, data). Строки if (!node.left) return node.right закрывают сразу случаи 1 и 2: у листа оба ребёнка null, и вернётся null.

Функцию removeNode я вынес за класс, чтобы не появлялся лишний публичный метод. Можно сделать её приватным методом #removeNode, тестам это не важно.

Сложность: все операции идут по одной ветке, то есть O(h), где h — высота дерева. В сбалансированном дереве это O(log n). Но пример из README добавляет 1, 2, 3, 4, 5 по возрастанию, и дерево вытягивается в «палку» высотой n. Там каждая операция O(n). Подробнее в уроке о деревьях.
