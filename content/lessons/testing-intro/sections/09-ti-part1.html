Задание Testing Intro. Часть 1: тесты каталога

О задании

Есть мини-магазин из двух страниц: каталог и корзина. Интерфейс уже готов, проверять его не нужно, тесты работают только с функциями бизнес-логики. Максимум 100 баллов:

ЧастьЧто делатьБаллы
1. КаталогФункции готовы. Написать 36 тестов для четырёх функций.40
2. КорзинаТесты готовы. Написать пять функций, чтобы тесты стали зелёными.40
ОформлениеФорк, ветка, конвенция коммитов, PR, два скриншота.20

Штраф −40: изменить src/catalog.js или src/data/products.js. Ещё −40: изменить tests/cart/cart.test.js, package.json или настройки Jest. В первой части меняете только файлы в tests/catalog/, во второй — только src/cart.js.

Цены хранятся в копейках: 49900 — это 499 ₽. Причина в том, что дробные числа в JavaScript считаются неточно: 0.1 + 0.2 даёт 0.30000000000000004, а 499.9 * 3 — 1499.6999999999998. Целые копейки считаются точно: 49900 * 3 = 149700.

Задание просит сравнивать массивы ID, а не объекты целиком. Так тесты короче и понятнее. Для этого в начале каждого файла заведём маленький помощник:

filterByCategory — 9 тестов

Сама функция (её не меняем) выглядит так:

function filterByCategory(products, category) {
  if (!category || category === 'all') return [...products];
  return products.filter((product) => product.category === category);
}

!category верно и для undefined, и для пустой строки, поэтому все три первых случая возвращают весь каталог. Самый прямолинейный вариант — по одному test на строку таблицы, заменяя каждый test.todo:

const products = require('../../src/data/products');
const { filterByCategory } = require('../../src/catalog');

const ids = (list) => list.map((product) => product.id);

describe('filterByCategory', () => {
  test("'all' возвращает товары с ID [1, 2, 3, 4, 5, 6]", () => {
    expect(ids(filterByCategory(products, 'all'))).toEqual([1, 2, 3, 4, 5, 6]);
  });

  test('undefined возвращает товары с ID [1, 2, 3, 4, 5, 6]', () => {
    expect(ids(filterByCategory(products, undefined))).toEqual([1, 2, 3, 4, 5, 6]);
  });

  test('пустая строка возвращает товары с ID [1, 2, 3, 4, 5, 6]', () => {
    expect(ids(filterByCategory(products, ''))).toEqual([1, 2, 3, 4, 5, 6]);
  });

  test("'electronics' возвращает товары с ID [1, 2]", () => {
    expect(ids(filterByCategory(products, 'electronics'))).toEqual([1, 2]);
  });

  test("'books' возвращает товары с ID [3, 6]", () => {
    expect(ids(filterByCategory(products, 'books'))).toEqual([3, 6]);
  });

  test("'home' возвращает товары с ID [4, 5]", () => {
    expect(ids(filterByCategory(products, 'home'))).toEqual([4, 5]);
  });

  test("'clothes' возвращает пустой массив", () => {
    expect(filterByCategory(products, 'clothes')).toEqual([]);
  });

  test("'toys' возвращает пустой массив", () => {
    expect(filterByCategory(products, 'toys')).toEqual([]);
  });

  test("'sports' возвращает пустой массив", () => {
    expect(filterByCategory(products, 'sports')).toEqual([]);
  });
});

searchProducts — 9 тестов и test.each

В прошлом файле много почти одинаковых тестов. test.each убирает повторы: передаём таблицу данных, и Jest создаёт по тесту на каждую строку. В названии %s заменяется первым значением строки, %j — им же, но в формате JSON (так видно невидимые символы вроде \t).

const products = require('../../src/data/products');
const { searchProducts } = require('../../src/catalog');

const ids = (list) => list.map((product) => product.id);

describe('searchProducts', () => {
  describe('точное совпадение', () => {
    test.each([
      ['Наушники', [1]],
      ['Чистый код', [6]],
      ['Настольная лампа', [4]],
    ])('запрос %s находит нужный товар', (query, expected) => {
      expect(ids(searchProducts(products, query))).toEqual(expected);
    });
  });

  describe('регистр не влияет на поиск', () => {
    test.each([
      ['javascript', [3]],
      ['ЧЕХОЛ', [5]],
      ['наушники', [1]],
    ])('запрос %s находит товар без учёта регистра', (query, expected) => {
      expect(ids(searchProducts(products, query))).toEqual(expected);
    });
  });

  describe('пустой запрос', () => {
    test.each(['', '   ', '\t\n'])('запрос %j возвращает весь каталог', (query) => {
      expect(ids(searchProducts(products, query))).toEqual([1, 2, 3, 4, 5, 6]);
    });
  });
});

Каждая строка внутри test.each([...]) — отдельный тест, их здесь 3 + 3 + 3 = 9. Значения строки приходят в функцию теста как аргументы: (query, expected). Вложенные describe повторяют три сценария из таблицы задания, так проверяющему легче сверять. '\t\n' — табуляция и перенос строки: trim() внутри функции их убирает, и запрос считается пустым.

sortProductsByPrice — 9 тестов

Здесь две особенности. Нужны массивы из отдельных товаров в заданном порядке, и нужно проверить, что функция не меняет исходный массив. Для первого пишем помощник pick: по списку ID собирает товары именно в таком порядке.

const products = require('../../src/data/products');
const { sortProductsByPrice } = require('../../src/catalog');

const ids = (list) => list.map((product) => product.id);
const pick = (order) => order.map((id) => products.find((product) => product.id === id));

describe('sortProductsByPrice', () => {
  describe('по возрастанию', () => {
    test("все товары с 'asc'", () => {
      expect(ids(sortProductsByPrice(products, 'asc'))).toEqual([3, 6, 5, 4, 2, 1]);
    });

    test('все товары без направления', () => {
      expect(ids(sortProductsByPrice(products))).toEqual([3, 6, 5, 4, 2, 1]);
    });

    test("товары [1, 3, 4] с 'asc'", () => {
      expect(ids(sortProductsByPrice(pick([1, 3, 4]), 'asc'))).toEqual([3, 4, 1]);
    });
  });

  describe('по убыванию', () => {
    test("все товары с 'desc'", () => {
      expect(ids(sortProductsByPrice(products, 'desc'))).toEqual([1, 2, 4, 5, 6, 3]);
    });

    test("товары [3, 1, 4] с 'desc'", () => {
      expect(ids(sortProductsByPrice(pick([3, 1, 4]), 'desc'))).toEqual([1, 4, 3]);
    });

    test("товары [5, 2, 6] с 'desc'", () => {
      expect(ids(sortProductsByPrice(pick([5, 2, 6]), 'desc'))).toEqual([2, 5, 6]);
    });
  });

  describe('исходный массив не меняется', () => {
    test("после сортировки [1, 3, 4] с 'asc'", () => {
      const list = pick([1, 3, 4]);
      sortProductsByPrice(list, 'asc');
      expect(ids(list)).toEqual([1, 3, 4]);
    });

    test("после сортировки [1, 3, 4] с 'desc'", () => {
      const list = pick([1, 3, 4]);
      sortProductsByPrice(list, 'desc');
      expect(ids(list)).toEqual([1, 3, 4]);
    });

    test('после сортировки всего каталога без направления', () => {
      sortProductsByPrice(products);
      expect(ids(products)).toEqual([1, 2, 3, 4, 5, 6]);
    });
  });
});

Задание предлагает собирать массивы через products.filter(({ id }) => [1, 3, 4].includes(id)). Это работает для [1, 3, 4], но filter сохраняет порядок каталога, поэтому для [3, 1, 4] и [5, 2, 6] получится [1, 3, 4] и [2, 5, 6], а не указанный порядок. На результат сортировки это не влияет, но pick честнее повторяет условие.

Зачем тесты «не меняется»? Внутри функции [...products].sort(...): сначала копия, потом сортировка. Метод sort меняет массив, на котором вызван. Я проверил: если убрать копию и написать products.sort(...), остальные 6 тестов остаются зелёными, а падают ровно эти 3. Без них ошибку бы никто не заметил, а сортировка в каталоге портила бы порядок исходного массива.

getProductById — 9 тестов

const products = require('../../src/data/products');
const { getProductById } = require('../../src/catalog');

describe('getProductById', () => {
  describe('товар существует', () => {
    test.each([
      [1, 0],
      [4, 3],
      [6, 5],
    ])('ID %s возвращает нужный объект', (id, index) => {
      expect(getProductById(products, id)).toBe(products[index]);
    });
  });

  describe('числовой ID отсутствует', () => {
    test.each([0, 999, -1])('ID %s возвращает null', (id) => {
      expect(getProductById(products, id)).toBeNull();
    });
  });

  describe('ID неверного типа', () => {
    test.each(['1', undefined, NaN])('ID %p возвращает null', (id) => {
      expect(getProductById(products, id)).toBeNull();
    });
  });
});
  • Для найденного товара используем toBe: функция возвращает тот самый объект из каталога (find не копирует), и задание формулирует ожидание как «объект products[0]». toEqual тоже прошёл бы.
  • Если в test.each передан просто массив значений, а не массив массивов, каждое значение приходит в тест первым аргументом.
  • '1' не находится, потому что внутри ===: строка '1' не равна числу 1. NaN не равен вообще ничему, даже самому себе.
  • %p выводит значение так, чтобы было видно тип: '1' в кавычках, undefined, NaN.
