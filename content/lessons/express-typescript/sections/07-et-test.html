Тесты: unit для логики, supertest для HTTP

Unit-тесты чистых функций

// src/domain/cart.ts
export interface CartLine {
  productId: string;
  price: number;
  quantity: number;
}

export interface CartTotals {
  subtotal: number;
  discount: number;
  total: number;
}

const PROMO_CODES: Readonly<Record<string, number>> = { SALE10: 10 };

export function addToCart(cart: readonly CartLine[], item: Omit<CartLine, 'quantity'>): CartLine[] {
  const existing = cart.find((line) => line.productId === item.productId);
  if (!existing) return [...cart, { ...item, quantity: 1 }];
  return cart.map((line) => (line === existing ? { ...line, quantity: line.quantity + 1 } : line));
}

export function calculateCart(cart: readonly CartLine[], promo?: string): CartTotals {
  const subtotal = cart.reduce((sum, line) => sum + line.price * line.quantity, 0);
  const percent = promo === undefined ? 0 : (PROMO_CODES[promo.trim().toUpperCase()] ?? 0);
  const discount = Math.round((subtotal * percent) / 100);
  return { subtotal, discount, total: subtotal - discount };
}
// tests/cart.test.ts
import { describe, expect, it } from 'vitest';
import { addToCart, calculateCart, type CartLine } from '../src/domain/cart.js';

const book = { productId: 'b1', price: 150000 };

describe('calculateCart', () => {
  it('складывает цену × количество по всем строкам', () => {
    const cart: CartLine[] = [{ ...book, quantity: 2 }, { productId: 'g1', price: 249000, quantity: 1 }];
    expect(calculateCart(cart)).toEqual({ subtotal: 549000, discount: 0, total: 549000 });
  });

  it('пустая корзина даёт нули', () => {
    expect(calculateCart([])).toEqual({ subtotal: 0, discount: 0, total: 0 });
  });

  it('SALE10 даёт 10%, регистр и пробелы не важны', () => {
    expect(calculateCart([{ ...book, quantity: 1 }], ' sale10 ')).toEqual({ subtotal: 150000, discount: 15000, total: 135000 });
  });

  it('неизвестный промокод не даёт скидки', () => {
    expect(calculateCart([{ ...book, quantity: 1 }], 'FREE').discount).toBe(0);
  });

  it('повторный расчёт не накапливает скидку', () => {
    const cart = [{ ...book, quantity: 1 }];
    calculateCart(cart, 'SALE10');
    expect(calculateCart(cart, 'SALE10').total).toBe(135000);
  });
});

describe('addToCart', () => {
  it('повторное добавление увеличивает количество, а не добавляет строку', () => {
    const once = addToCart([], book);
    const twice = addToCart(once, book);
    expect(twice).toEqual([{ ...book, quantity: 2 }]);
    expect(once).toEqual([{ ...book, quantity: 1 }]);
  });
});

Каждый тест проверяет одно правило, и его название читается как требование. Сценарии выбраны по классам эквивалентности и границам (как их выбирать): несколько строк, пустая корзина, верный код в другом регистре, неверный код, повторный расчёт.

Задача 15. Unit-тесты расчёта корзины ★☆☆ разминка

Не подглядывая в tests/cart.test.ts из урока, напишите свои unit-тесты для calculateCart: минимум пять сценариев (обычный, пустая корзина, промокод в другом регистре, неизвестный промокод, округление). Проверьте их мутантом: замените Math.round на Math.floor — хотя бы один тест должен упасть.

// cart.ts
export interface CartLine {
  productId: string;
  price: number;
  quantity: number;
}

const PROMO_CODES: Readonly<Record<string, number>> = { SALE10: 10 };

export function calculateCart(cart: readonly CartLine[], promo?: string) {
  const subtotal = cart.reduce((sum, line) => sum + line.price * line.quantity, 0);
  const percent = promo === undefined ? 0 : (PROMO_CODES[promo.trim().toUpperCase()] ?? 0);
  const discount = Math.round((subtotal * percent) / 100);
  return { subtotal, discount, total: subtotal - discount };
}

Зачем: тест на округление — тот самый, что разбирается в разборе live coding. Если вы напишете его сами, на интервью это будет не заученный ответ, а свой опыт.

Решение и разбор
// cart.test.ts
import { describe, expect, it } from 'vitest';
import { calculateCart } from './cart.js';

describe('calculateCart', () => {
  it('сумма с количеством', () => {
    expect(calculateCart([{ productId: 'a', price: 100, quantity: 3 }])).toEqual({ subtotal: 300, discount: 0, total: 300 });
  });
  it('пустая корзина', () => {
    expect(calculateCart([])).toEqual({ subtotal: 0, discount: 0, total: 0 });
  });
  it('промокод без учёта регистра и пробелов', () => {
    expect(calculateCart([{ productId: 'a', price: 1000, quantity: 1 }], ' sale10 ').discount).toBe(100);
  });
  it('неизвестный промокод', () => {
    expect(calculateCart([{ productId: 'a', price: 1000, quantity: 1 }], 'FREE').discount).toBe(0);
  });
  it('скидка округляется математически: 9 999,9 → 10 000', () => {
    expect(calculateCart([{ productId: 'a', price: 99999, quantity: 1 }], 'SALE10').discount).toBe(10000);
  });
});

Мутант с Math.floor ловит только тест на округление: на «круглых» суммах round и floor совпадают. Поэтому такой тест не лишний, даже если кажется мелочью. Название теста фиксирует решение (математическое округление). Если бизнес решит иначе, тест покажет, где это правило живёт.

HTTP-тесты через supertest

supertest принимает приложение Express, сам поднимает его на временном порту, отправляет запрос и даёт проверить статус и тело. Это интеграционный тест: проверяется связка «маршрут + проверка + логика + обработчик ошибок» (уровни тестов).

// tests/api.test.ts
import request from 'supertest';
import { describe, expect, it } from 'vitest';
import { createApp } from '../src/app.js';

const app = createApp();

describe('GET /api/products', () => {
  it('фильтрует по поиску без учёта регистра', async () => {
    const res = await request(app).get('/api/products').query({ search: 'КОД' });
    expect(res.status).toBe(200);
    expect(res.body.map((p: { id: string }) => p.id)).toEqual(['b1']);
  });

  it('отвечает 400 на нечисловую цену', async () => {
    const res = await request(app).get('/api/products?minPrice=abc');
    expect(res.status).toBe(400);
    expect(res.body).toEqual({ error: 'minPrice должен быть целым числом ≥ 0' });
  });

  it('отвечает 404 на неизвестный id', async () => {
    const res = await request(app).get('/api/products/nope');
    expect(res.status).toBe(404);
  });
});

describe('POST /api/cart/quote', () => {
  it('считает итог со скидкой', async () => {
    const res = await request(app)
      .post('/api/cart/quote')
      .send({ items: [{ productId: 'b1', quantity: 2 }], promo: 'SALE10' });
    expect(res.body).toEqual({ subtotal: 300000, discount: 30000, total: 270000 });
  });

  it('отвечает 400 на кривое тело', async () => {
    const res = await request(app).post('/api/cart/quote').send({ items: [{ productId: 'b1', quantity: 0 }] });
    expect(res.status).toBe(400);
  });

  it('отвечает 400 на битый JSON', async () => {
    const res = await request(app).post('/api/cart/quote').set('Content-Type', 'application/json').send('{oops');
    expect(res.status).toBe(400);
  });
});

it('неизвестный маршрут — 404 в JSON', async () => {
  const res = await request(app).get('/api/nope');
  expect(res.status).toBe(404);
  expect(res.body).toEqual({ error: 'Маршрут не найден' });
});

Задача 16. HTTP-тесты ошибок ★★☆ практика

Для маршрута создания товара из задачи «Создание товара с проверкой тела» напишите HTTP-тесты на supertest. Пустое тело, тело-массив, отсутствие заголовка Content-Type и битый JSON дают 400, а не 500. Если какой-то тест упадёт, исправьте обработку в приложении, а не тест.

// app.ts
import express, { type ErrorRequestHandler } from 'express';

export function createApp() {
  const app = express();
  app.use(express.json());
  app.post('/api/products', (req, res) => {
    const { title, price } = req.body ?? {};
    if (typeof title !== 'string' || title.trim() === '' || !Number.isInteger(price) || price < 0) {
      return res.status(400).json({ error: 'Неверные данные' });
    }
    res.status(201).json({ id: '1', title: title.trim(), price });
  });
  const onError: ErrorRequestHandler = (err, _req, res, _next) => {
    const status = err.status ?? 500;
    res.status(status).json({ error: status >= 500 ? 'Внутренняя ошибка сервера' : err.message });
  };
  app.use(onError);
  return app;
}

Зачем: счастливый путь вы проверите руками в браузере, а кривые запросы руками отправляют редко. Именно там прячутся ответы 500, после которых клиент думает, что упал сервер.

Решение и разбор
// app.test.ts
import request from 'supertest';
import { describe, expect, it } from 'vitest';
import { createApp } from './app.js';

const post = () => request(createApp()).post('/api/products');

describe('POST /api/products: кривые запросы', () => {
  it('без тела — 400', async () => {
    expect((await post()).status).toBe(400);
  });
  it('тело-массив — 400', async () => {
    expect((await post().send([1, 2])).status).toBe(400);
  });
  it('без Content-Type — 400', async () => {
    expect((await post().set('Content-Type', 'text/plain').send('title=x')).status).toBe(400);
  });
  it('битый JSON — 400 с сообщением', async () => {
    const res = await post().set('Content-Type', 'application/json').send('{oops');
    expect(res.status).toBe(400);
    expect(res.body.error).toBeTypeOf('string');
  });
  it('правильное тело — 201', async () => {
    expect((await post().send({ title: 'Лампа', price: 27900 })).status).toBe(201);
  });
});

Без тела и без JSON-заголовка req.body равен undefined. Деструктуризация const { title } = req.body бросила бы TypeError, и клиент получил бы 500. Защита — req.body ?? {}. Мутант без неё и с обработчиком, который всё превращает в 500, наши тесты ловят.
