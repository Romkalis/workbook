Express

Express — минималистичный и самый популярный веб-фреймворк для Node.js. Он даёт три вещи: маршрутизацию (routing: какой код вызвать на GET /api/todos/5), удобные req и res и цепочку middleware. Установка: npm install express — сейчас ставится Express 5, ему нужен Node.js 18+.

Первое приложение: API списка задач

const express = require('express');

const app = express();
app.use(express.json()); // разбирать JSON в теле запроса

let todos = [{ id: 1, title: 'Выучить Node.js', done: false }];
let nextId = 2;

app.get('/api/todos', (req, res) => {
  const { done } = req.query;
  const result = done === undefined
    ? todos
    : todos.filter((t) => String(t.done) === done);
  res.json(result);
});

app.get('/api/todos/:id', (req, res) => {
  const todo = todos.find((t) => t.id === Number(req.params.id));
  if (!todo) return res.status(404).json({ error: 'Задача не найдена' });
  res.json(todo);
});

app.post('/api/todos', (req, res) => {
  const { title } = req.body ?? {};
  if (typeof title !== 'string' || title.trim() === '') {
    return res.status(400).json({ error: 'Нужно поле title' });
  }
  const todo = { id: nextId++, title: title.trim(), done: false };
  todos.push(todo);
  res.status(201).json(todo);
});

app.patch('/api/todos/:id', (req, res) => {
  const todo = todos.find((t) => t.id === Number(req.params.id));
  if (!todo) return res.status(404).json({ error: 'Задача не найдена' });
  if (typeof req.body?.done === 'boolean') todo.done = req.body.done;
  res.json(todo);
});

app.delete('/api/todos/:id', (req, res) => {
  todos = todos.filter((t) => t.id !== Number(req.params.id));
  res.status(204).end();
});

app.listen(3002, () => console.log('http://localhost:3002'));

Задачи хранятся в массиве, поэтому пропадут при перезапуске — базу данных подключим в следующем разделе. Проверено curl:

ЗапросОтвет
GET /api/todos200, массив с одной задачей
POST /api/todos с {"title":" Написать сервер "}201, {"id":2,"title":"Написать сервер","done":false}
POST /api/todos с {} или без тела400, {"error":"Нужно поле title"}
PATCH /api/todos/2 с {"done":true}200, задача с done: true
GET /api/todos?done=true200, только выполненные
GET /api/todos/99404
DELETE /api/todos/1204 без тела

Это REST-API из урока о клиенте и сервере: ресурс /api/todos, действие выражено методом, ответ — правильным кодом состояния.

req и res

Где брать данные из запроса (гайд по маршрутизации):

  • req.params — части пути из шаблона. Для маршрута /api/todos/:id и адреса /api/todos/5 будет { id: '5' }. Всегда строки, поэтому Number(...).
  • req.query — параметры после ?: ?done=true даёт { done: 'true' }, тоже строкой.
  • req.body — тело запроса. Появляется, только если подключён разборщик: express.json() для JSON. Если тела нет, в Express 5 req.body равен undefined, поэтому в примере req.body ?? {}.
  • req.headers, req.get('Authorization') — заголовки.

Как ответить: res.json(obj) (сам ставит Content-Type: application/json), res.send('текст'), res.status(201) (возвращает res, поэтому пишут цепочкой res.status(201).json(...)), res.status(204).end(), res.redirect('/login'), res.cookie(...).

На один запрос — один ответ. Частая ошибка: забыть return после res.status(404).json(...); код пойдёт дальше, попытается ответить второй раз, и Express упадёт с «Cannot set headers after they are sent to the client».

Middleware

Middleware («промежуточный обработчик», читается «мидлвэр») — функция (req, res, next), через которую проходит запрос по пути к маршруту. Она может что-то сделать с запросом (разобрать тело, проверить токен, записать в лог) и либо ответить сама, либо вызвать next(), чтобы передать запрос дальше (гайд).

app.use((req, res, next) => {
  console.log(req.method, req.url);
  next();
});

Middleware можно подключить ко всему приложению (app.use(fn)), к пути (app.use('/api', fn)) или к одному маршруту (app.get('/me', requireAuth, handler)) — именно так в уроке о JWT защищён профиль. Готовые middleware: express.json(), express.static('public') (раздаёт файлы из папки), cors() из пакета cors (выставляет заголовки CORS), cookie-parser.

404 и обработка ошибок

app.get('/api/boom', async () => {
  throw new Error('Что-то сломалось');
});

// никто не ответил — 404
app.use((req, res) => {
  res.status(404).json({ error: 'Маршрут не найден' });
});

// обработчик ошибок: ровно четыре параметра
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: 'Внутренняя ошибка сервера' });
});

Обработчик ошибок Express узнаёт по четырём параметрам, даже если next не используется. Подключается он последним. Проверка: /api/boom вернул 500 с сообщением для клиента, а текст настоящей ошибки остался в консоли сервера; /nope вернул 404. Наружу не отдают стек и подробности — это граница ошибок из урока об обработке ошибок (гайд Express).

Express 4 и Express 5

В примере ошибка выброшена в async-обработчике и всё равно попала в обработчик ошибок. Это работает только в Express 5: он сам ловит отклонённые промисы. В Express 4, который всё ещё встречается в туториалах, такая ошибка повесила бы запрос; там нужно оборачивать код в try...catch и вызывать next(error). Другие отличия 5-й версии — в руководстве по миграции.

Router: маршруты по файлам

Когда маршрутов много, их выносят в отдельные файлы через express.Router():

// routes/todos.js
const { Router } = require('express');
const router = Router();
router.get('/', listTodos);
router.post('/', createTodo);
module.exports = router;

// app.js
app.use('/api/todos', require('./routes/todos'));

Пути внутри роутера пишутся относительно префикса: router.get('/') отвечает на GET /api/todos.
