Express

Express — минималистичный и самый популярный веб-фреймворк для Node.js. Он даёт три вещи: маршрутизацию (routing: какой код вызвать на GET /api/todos/5), удобные req и res и цепочку middleware. Установка: npm install express — сейчас ставится Express 5, ему нужен Node.js 18+.

Первое приложение: API списка задач

const express = require('express');

const app = express();
app.use(express.json()); // разбирать JSON в теле запроса

let todos = [{ id: 1, title: 'Выучить Node.js', done: false }];
let nextId = 2;

app.get('/api/todos', (req, res) => {
  const { done } = req.query;
  const result = done === undefined
    ? todos
    : todos.filter((t) => String(t.done) === done);
  res.json(result);
});

app.get('/api/todos/:id', (req, res) => {
  const todo = todos.find((t) => t.id === Number(req.params.id));
  if (!todo) return res.status(404).json({ error: 'Задача не найдена' });
  res.json(todo);
});

app.post('/api/todos', (req, res) => {
  const { title } = req.body ?? {};
  if (typeof title !== 'string' || title.trim() === '') {
    return res.status(400).json({ error: 'Нужно поле title' });
  }
  const todo = { id: nextId++, title: title.trim(), done: false };
  todos.push(todo);
  res.status(201).json(todo);
});

app.patch('/api/todos/:id', (req, res) => {
  const todo = todos.find((t) => t.id === Number(req.params.id));
  if (!todo) return res.status(404).json({ error: 'Задача не найдена' });
  if (typeof req.body?.done === 'boolean') todo.done = req.body.done;
  res.json(todo);
});

app.delete('/api/todos/:id', (req, res) => {
  todos = todos.filter((t) => t.id !== Number(req.params.id));
  res.status(204).end();
});

app.listen(3002, () => console.log('http://localhost:3002'));

Задачи хранятся в массиве, поэтому пропадут при перезапуске — базу данных подключим в следующем разделе. Проверено curl:

ЗапросОтвет
GET /api/todos200, массив с одной задачей
POST /api/todos с {"title":" Написать сервер "}201, {"id":2,"title":"Написать сервер","done":false}
POST /api/todos с {} или без тела400, {"error":"Нужно поле title"}
PATCH /api/todos/2 с {"done":true}200, задача с done: true
GET /api/todos?done=true200, только выполненные
GET /api/todos/99404
DELETE /api/todos/1204 без тела

Это REST-API из урока о клиенте и сервере: ресурс /api/todos, действие выражено методом, ответ — правильным кодом состояния.

Задача 22. Тот же каталог на Express ★☆☆ разминка

Перепишите каталог на Express: createApp(products) возвращает приложение с маршрутами GET /api/products (фильтр ?category=) и GET /api/products/:id (404, если нет). Сравните объём кода с вариантом на http.

Зачем: сравнение показывает, за что отвечает фреймворк: разбор пути и параметров, res.json, маршрутизацию по методу. Логика (что отдать) осталась вашей, а «водопровод» ушёл.

Решение и разбор
// app.mjs
import express from 'express';

export function createApp(products) {
  const app = express();

  app.get('/api/products', (req, res) => {
    const { category } = req.query;
    res.json(category ? products.filter((p) => p.category === category) : products);
  });

  app.get('/api/products/:id', (req, res) => {
    const product = products.find((p) => p.id === Number(req.params.id));
    if (!product) return res.status(404).json({ error: 'Товар не найден' });
    res.json(product);
  });

  return app;
}

req.params.id — строка, отсюда Number. Без него '1' === 1 ложно, и ни один товар не найдётся. return перед ответом 404 обязателен, следующая задача как раз об этом. Приложение возвращается без listen, по тем же причинам, что и раньше.

req и res

Где брать данные из запроса (гайд по маршрутизации):

  • req.params — части пути из шаблона. Для маршрута /api/todos/:id и адреса /api/todos/5 будет { id: '5' }. Всегда строки, поэтому Number(...).
  • req.query — параметры после ?: ?done=true даёт { done: 'true' }, тоже строкой.
  • req.body — тело запроса. Появляется, только если подключён разборщик: express.json() для JSON. Если тела нет, в Express 5 req.body равен undefined, поэтому в примере req.body ?? {}.
  • req.headers, req.get('Authorization') — заголовки.

Как ответить: res.json(obj) (сам ставит Content-Type: application/json), res.send('текст'), res.status(201) (возвращает res, поэтому пишут цепочкой res.status(201).json(...)), res.status(204).end(), res.redirect('/login'), res.cookie(...).

На один запрос — один ответ. Частая ошибка: забыть return после res.status(404).json(...); код пойдёт дальше, попытается ответить второй раз, и Express упадёт с «Cannot set headers after they are sent to the client».

Задача 23. «Cannot set headers after they are sent» ★☆☆ разминка

Маршрут ниже при несуществующем id роняет обработку с этой ошибкой в консоли. Найдите причину и исправьте. Затем переделайте маршрут, чтобы ошибка была невозможна в принципе, без return перед ответом.

app.get('/api/products/:id', (req, res) => {
  const product = products.find((p) => p.id === Number(req.params.id));
  if (!product) {
    res.status(404).json({ error: 'Товар не найден' });
  }
  res.json(product);
});

Зачем: правило «один запрос — один ответ» нарушается молча и проявляется ошибкой в логах. Код, где все ветки явно заканчиваются ответом, читается и проверяется легче, чем код с «забытыми» продолжениями.

Решение и разбор
// app.mjs
import express from 'express';

export function createApp(products) {
  const app = express();
  app.get('/api/products/:id', (req, res) => {
    const product = products.find((p) => p.id === Number(req.params.id));
    if (product) res.json(product);
    else res.status(404).json({ error: 'Товар не найден' });
  });
  return app;
}

Причина: после ответа 404 код шёл дальше и вызывал res.json(undefined), то есть второй ответ на тот же запрос. Короткое исправление — return res.status(404).... Вариант с if/else делает ветки взаимоисключающими по структуре. Оба приемлемы, главное — держаться одного стиля во всём проекте.

Middleware

Middleware («промежуточный обработчик», читается «мидлвэр») — функция (req, res, next), через которую проходит запрос по пути к маршруту. Она может что-то сделать с запросом (разобрать тело, проверить токен, записать в лог) и либо ответить сама, либо вызвать next(), чтобы передать запрос дальше (гайд).

app.use((req, res, next) => {
  console.log(req.method, req.url);
  next();
});

Middleware можно подключить ко всему приложению (app.use(fn)), к пути (app.use('/api', fn)) или к одному маршруту (app.get('/me', requireAuth, handler)) — именно так в уроке о JWT защищён профиль. Готовые middleware: express.json(), express.static('public') (раздаёт файлы из папки), cors() из пакета cors (выставляет заголовки CORS), cookie-parser.

Иногда middleware нужно сделать что-то после ответа: записать в лог статус и время обработки. В момент вызова next() ответ ещё не отправлен, поэтому подписываются на событие ответа finish. Оно срабатывает, когда ответ полностью отдан:

app.use((req, res, next) => {
  const start = performance.now();
  res.on('finish', () => {
    console.log(req.method, req.originalUrl, res.statusCode, Math.round(performance.now() - start), 'мс');
  });
  next();
});

Задача 24. Middleware: время ответа ★★☆ практика

Упражнение из урока: middleware timing(log), которое после отправки ответа вызывает log со строкой вида GET /api/products 200 — 3 мс. Функция log передаётся параметром (по умолчанию console.log), чтобы её можно было подменить в тесте.

Зачем: middleware — место для того, что нужно всем маршрутам: лог, время, заголовки. Время меряют по событию finish: в момент вызова next() ответ ещё не отправлен, и замер получится нулевым.

Решение и разбор
// timing.mjs
export function timing(log = console.log) {
  return (req, res, next) => {
    const start = performance.now();
    res.on('finish', () => {
      const ms = Math.round(performance.now() - start);
      log(`${req.method} ${req.originalUrl} ${res.statusCode} — ${ms} мс`);
    });
    next();
  };
}

Middleware сразу вызывает next(), не задерживая запрос, а запись делает позже, когда сработает finish. req.originalUrl, а не req.url: внутри роутера req.url уже без префикса. performance.now() точнее и надёжнее Date.now() для измерения длительности. Фабрика timing(log), а не готовая функция, позволяет настроить поведение: тот же приём, что у express.json(options).

Задача 25. Middleware для одного маршрута ★★☆ практика

Напишите requireApiKey(expected): пропускает запрос, если заголовок X-Api-Key равен ключу, иначе отвечает 401. Подключите её только к DELETE /api/products/:id; чтение оставьте открытым.

Зачем: это та же схема, по которой в уроке о JWT защищают профиль: проверка как отдельная функция в цепочке маршрута. Маршрут удаления остаётся простым, а правило доступа видно в одной строке подключения.

Решение и разбор
// app.mjs
import express from 'express';

export function requireApiKey(expected) {
  return (req, res, next) => {
    if (req.get('X-Api-Key') !== expected) {
      return res.status(401).json({ error: 'Нужен ключ API' });
    }
    next();
  };
}

export function createApp(products, apiKey) {
  const app = express();
  app.get('/api/products', (req, res) => res.json(products));
  app.delete('/api/products/:id', requireApiKey(apiKey), (req, res) => {
    const index = products.findIndex((p) => p.id === Number(req.params.id));
    if (index === -1) return res.status(404).json({ error: 'Товар не найден' });
    products.splice(index, 1);
    res.status(204).end();
  });
  return app;
}

401 означает «не представились» (нет или неверный ключ), 403 — «представились, но нельзя». Ключ передаётся в фабрику, а не читается внутри из process.env: так middleware можно протестировать и переиспользовать. Учтите: ключ в заголовке подходит для общения сервера с сервером. Во фронтенде любой ключ виден пользователю, поэтому там нужна авторизация пользователей (урок о JWT).

404 и обработка ошибок

app.get('/api/boom', async () => {
  throw new Error('Что-то сломалось');
});

// никто не ответил — 404
app.use((req, res) => {
  res.status(404).json({ error: 'Маршрут не найден' });
});

// обработчик ошибок: ровно четыре параметра
app.use((err, req, res, next) => {
  console.error(err.message);
  res.status(500).json({ error: 'Внутренняя ошибка сервера' });
});

Обработчик ошибок Express узнаёт по четырём параметрам, даже если next не используется. Подключается он последним. Проверка: /api/boom вернул 500 с сообщением для клиента, а текст настоящей ошибки остался в консоли сервера; /nope вернул 404. Наружу не отдают стек и подробности — это граница ошибок из урока об обработке ошибок (гайд Express).

Express 4 и Express 5

В примере ошибка выброшена в async-обработчике и всё равно попала в обработчик ошибок. Это работает только в Express 5: он сам ловит отклонённые промисы. В Express 4, который всё ещё встречается в туториалах, такая ошибка повесила бы запрос; там нужно оборачивать код в try...catch и вызывать next(error). Другие отличия 5-й версии — в руководстве по миграции.

Задача 26. Свои ошибки и общий обработчик ★★☆ практика

Создайте класс NotFoundError (статус 404) и общий обработчик ошибок, который превращает её в JSON-ответ 404, а любую другую ошибку — в 500 с общим текстом без подробностей. Маршрут GET /api/products/:id асинхронный: он получает товар через await repo.findById(id) и бросает NotFoundError, если товара нет.

Зачем: маршруты перестают решать, как отформатировать ошибку: они только сообщают, что случилось. Формат ответа и правило «наружу не отдаём подробностей» живут в одном месте, и поменять их можно одной правкой.

Решение и разбор
// app.mjs
import express from 'express';

export class NotFoundError extends Error {
  status = 404;
}

export function createApp(repo, log = console.error) {
  const app = express();

  app.get('/api/products/:id', async (req, res) => {
    const product = await repo.findById(Number(req.params.id));
    if (!product) throw new NotFoundError('Товар не найден');
    res.json(product);
  });

  app.use((err, req, res, next) => {
    if (err instanceof NotFoundError) return res.status(404).json({ error: err.message });
    log(err);
    res.status(500).json({ error: 'Внутренняя ошибка сервера' });
  });

  return app;
}

Исключение из async-маршрута Express 5 сам передаёт обработчику с четырьмя параметрами. Неизвестная ошибка логируется (через подменяемый log), а клиент видит общий текст: сообщение вроде «connect ECONNREFUSED 10.0.0.5:5432» рассказало бы постороннему об устройстве системы. Обработчик подключён последним, после всех маршрутов.

Router: маршруты по файлам

Когда маршрутов много, их выносят в отдельные файлы через express.Router():

// routes/todos.js
const { Router } = require('express');
const router = Router();
router.get('/', listTodos);
router.post('/', createTodo);
module.exports = router;

// app.js
app.use('/api/todos', require('./routes/todos'));

Пути внутри роутера пишутся относительно префикса: router.get('/') отвечает на GET /api/todos.

Задача 27. Маршруты в отдельном файле ★☆☆ разминка

Вынесите маршруты товаров в routes/products.mjs через Router, а в app.mjs подключите его под префиксом /api/products. Роутер должен получать данные параметром (фабрика), а не импортировать их.

Зачем: файл маршрутов отвечает на вопрос «какие URL есть у ресурса». Префикс задаётся в одном месте, поэтому переезд на /api/v2/products — одна строка. Данные параметром позволяют подставить в тесте другой набор.

Решение и разбор
// routes/products.mjs
import { Router } from 'express';

export function productsRouter(products) {
  const router = Router();
  router.get('/', (req, res) => res.json(products));
  router.get('/:id', (req, res) => {
    const product = products.find((p) => p.id === Number(req.params.id));
    if (!product) return res.status(404).json({ error: 'Товар не найден' });
    res.json(product);
  });
  return router;
}
// app.mjs
import express from 'express';
import { productsRouter } from './routes/products.mjs';

export function createApp(products) {
  const app = express();
  app.use('/api/products', productsRouter(products));
  return app;
}

Внутри роутера пути без префикса: '/' и '/:id'. Роутер ничего не знает о том, куда его подключили, поэтому его легко перенести. Это шаг к структуре из раздела урока: routes, затем controllers, когда обработчики разрастутся.
