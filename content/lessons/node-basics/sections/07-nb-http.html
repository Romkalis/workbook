HTTP-сервер на чистом Node.js

Модуль http умеет принимать HTTP-запросы. Express построен поверх него, поэтому сначала посмотрим, что под капотом.

const http = require('node:http');

const server = http.createServer((req, res) => {
  if (req.method === 'GET' && req.url === '/') {
    res.writeHead(200, { 'Content-Type': 'text/plain; charset=utf-8' });
    res.end('Привет от Node.js');
    return;
  }
  res.writeHead(404, { 'Content-Type': 'application/json' });
  res.end(JSON.stringify({ error: 'Not Found' }));
});

server.listen(3001, () => console.log('http://localhost:3001'));
  • createServer принимает функцию, которая вызывается на каждый запрос.
  • req (request) — входящий запрос: req.method, req.url (путь вместе с query string), req.headers.
  • res (response) — ответ, который мы собираем: writeHead задаёт статус и заголовки, end отправляет тело и завершает ответ. Если забыть end, браузер будет ждать вечно.
  • listen(3001) начинает слушать порт. Если порт занят другой программой, будет ошибка EADDRINUSE.

Проверка curl: curl localhost:3001/ вернул «Привет от Node.js», а /x — HTTP/1.1 404 Not Found. Это тот же текстовый HTTP, что разобран в уроке о клиенте и сервере, только теперь мы на стороне сервера.

Писать на чистом http настоящий API неудобно: маршруты разбираются вручную через if, тело запроса приходит потоком кусков и его надо собирать и парсить самому, query string — разбирать через new URL(req.url, 'http://localhost'). Эту рутину берёт на себя Express.

Как выглядит чтение тела вручную. Запрос — поток (stream): событие data приносит очередной кусок байтов (Buffer), событие end сообщает, что кусков больше не будет. Склеивать нужно байты, а декодировать в строку один раз в конце: русская буква может оказаться разрезанной между кусками.

function readJsonBody(req) {
  return new Promise((resolve, reject) => {
    const chunks = [];
    req.on('data', (chunk) => chunks.push(chunk));
    req.on('end', () => {
      try {
        resolve(JSON.parse(Buffer.concat(chunks).toString('utf8')));
      } catch (error) {
        reject(error); // битый JSON — ошибка клиента, ответ 400
      }
    });
    req.on('error', reject);
  });
}

В настоящем сервере ещё ограничивают размер тела, иначе клиент может прислать гигабайт и занять всю память. express.json() делает всё это сам, по умолчанию с лимитом 100 КБ (из документации Express).

Задача 19. Каталог на чистом http ★★☆ практика

Напишите createCatalogServer(products) на node:http (без listen, вернуть сервер). GET /api/products отдаёт JSON со статусом 200. Тот же путь с другим методом — 405 и заголовок Allow: GET. Любой другой путь — 404 в JSON. Ответы с правильным Content-Type.

Зачем: до Express полезно один раз собрать ответ руками: статус, заголовки, тело, end. Тогда понятно, что именно делает за вас res.json. А 405 вместо 404 — честный ответ: путь есть, метод нет.

Решение и разбор
// server.mjs
import http from 'node:http';

function sendJson(res, status, data, headers = {}) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8', ...headers });
  res.end(JSON.stringify(data));
}

export function createCatalogServer(products) {
  return http.createServer((req, res) => {
    const { pathname } = new URL(req.url, 'http://localhost');
    if (pathname === '/api/products') {
      if (req.method !== 'GET') return sendJson(res, 405, { error: 'Метод не поддерживается' }, { Allow: 'GET' });
      return sendJson(res, 200, products);
    }
    sendJson(res, 404, { error: 'Не найдено' });
  });
}

Помощник sendJson убирает повторение «статус + заголовок + stringify + end». Именно так появляются фреймворки. Путь берётся из new URL(...).pathname, а не из req.url напрямую: иначе /api/products?x=1 не совпал бы со строкой. Сервер создаётся без listen: тест сам решает, на каком порту его поднять.

Задача 20. Фильтр из строки запроса ★★☆ практика

Добавьте в сервер из прошлой задачи фильтры: ?category=books и ?maxPrice=100000. Если maxPrice не число — 400 с понятной ошибкой. Разбор параметров вынесите в чистую функцию parseFilters(searchParams).

Зачем: query string — это строки от пользователя. Разбор и проверка в одном месте превращают их в надёжный объект фильтров, а сам маршрут остаётся коротким.

Решение и разбор
// filters.mjs
export function parseFilters(params) {
  const filters = {};
  const category = params.get('category');
  if (category) filters.category = category;
  const maxPrice = params.get('maxPrice');
  if (maxPrice !== null) {
    const n = Number(maxPrice);
    if (maxPrice.trim() === '' || !Number.isFinite(n)) throw new RangeError('maxPrice должен быть числом');
    filters.maxPrice = n;
  }
  return filters;
}

export function applyFilters(products, f) {
  return products.filter(
    (p) => (!f.category || p.category === f.category) && (f.maxPrice === undefined || p.price <= f.maxPrice),
  );
}
// server.mjs
import http from 'node:http';
import { parseFilters, applyFilters } from './filters.mjs';

function sendJson(res, status, data) {
  res.writeHead(status, { 'Content-Type': 'application/json; charset=utf-8' });
  res.end(JSON.stringify(data));
}

export function createCatalogServer(products) {
  return http.createServer((req, res) => {
    const url = new URL(req.url, 'http://localhost');
    if (url.pathname !== '/api/products') return sendJson(res, 404, { error: 'Не найдено' });
    try {
      sendJson(res, 200, applyFilters(products, parseFilters(url.searchParams)));
    } catch (error) {
      if (!(error instanceof RangeError)) throw error;
      sendJson(res, 400, { error: error.message });
    }
  });
}

URLSearchParams.get возвращает null, если параметра нет. Так отличаются «не передан» и «передан пустым»: ?maxPrice= — это ошибка, а не «без ограничения». Отдельно проверяется trim() === '', потому что Number('') равно 0, а это тихая ловушка. В catch обрабатывается только RangeError от разбора, остальное пробрасывается: не надо выдавать собственный баг за ошибку клиента.

Задача 21. Тело POST-запроса без Express ★★★ с подвохом

Напишите readJsonBody(req): собирает тело запроса из кусков и разбирает JSON. Если JSON битый — отклоняется ошибкой со status = 400. Если тело больше 1 МБ — со status = 413. Используйте её в маршруте POST /api/echo, который отвечает тем же объектом.

Зачем: так вы увидите, что делает express.json(): потоковое чтение, лимит размера, обработку ошибок. Лимит — это защита: без него клиент может отправить гигабайт и занять всю память сервера.

Решение и разбор
// body.mjs
const LIMIT = 1024 * 1024;

export function readJsonBody(req) {
  return new Promise((resolve, reject) => {
    const chunks = [];
    let size = 0;
    req.on('data', (chunk) => {
      size += chunk.length;
      if (size <= LIMIT) chunks.push(chunk); // сверх лимита не копим, только дочитываем
    });
    req.on('end', () => {
      if (size > LIMIT) {
        return reject(Object.assign(new Error('Слишком большое тело'), { status: 413 }));
      }
      try {
        resolve(JSON.parse(Buffer.concat(chunks).toString('utf8')));
      } catch {
        reject(Object.assign(new Error('Некорректный JSON'), { status: 400 }));
      }
    });
    req.on('error', reject);
  });
}
// server.mjs
import http from 'node:http';
import { readJsonBody } from './body.mjs';

export function createServer() {
  return http.createServer(async (req, res) => {
    if (req.method === 'POST' && req.url === '/api/echo') {
      try {
        const data = await readJsonBody(req);
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(data));
      } catch (error) {
        res.writeHead(error.status ?? 500, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ error: error.message }));
      }
      return;
    }
    res.writeHead(404).end();
  });
}

Запрос — поток: событие data приносит кусок байтов (Buffer), end сообщает, что кусков больше не будет. Склеиваются именно байты (Buffer.concat), а потом декодируются один раз. Если декодировать каждый кусок отдельно, русская буква, разрезанная между кусками, превратится в мусор. Сверх лимита куски не сохраняются, поэтому память не растёт, а поток дочитывается до конца, чтобы клиент получил честный ответ 413. Можно и сразу оборвать соединение (req.destroy()): это экономит трафик, но клиент увидит обрыв вместо понятного статуса. Выбор зависит от того, кто ваши клиенты.
