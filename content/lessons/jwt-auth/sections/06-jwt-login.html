Регистрация и вход

Регистрация

// routes/auth.js
const router = require('express').Router();
const jwt = require('jsonwebtoken');
const User = require('../models/User');

router.post('/register', async (req, res) => {
  try {
    const { email, password } = req.body;
    const user = await User.create({ email, password }); // хук pre('save') захеширует пароль
    res.status(201).json({ id: user._id, email: user.email });
  } catch (err) {
    res.status(400).json({ message: err.message });
  }
});

Токен здесь ещё не создаётся. Мы просто сохраняем пользователя. Обратите внимание: в ответе нет пароля, даже хешированного.

Вход — здесь появляется токен

router.post('/login', async (req, res) => {
  try {
    const { email, password } = req.body;

    const user = await User.findOne({ email });
    if (!user) return res.status(401).json({ message: 'Неверный email или пароль' });

    const isMatch = await user.comparePassword(password);
    if (!isMatch) return res.status(401).json({ message: 'Неверный email или пароль' });

    const token = jwt.sign(
      { id: user._id },                          // payload: что положить в токен
      process.env.JWT_SECRET,                    // секрет для подписи
      { expiresIn: process.env.JWT_EXPIRES_IN }, // срок жизни
    );
    res.json({ token });
  } catch (err) {
    res.status(500).json({ message: 'Ошибка сервера' });
  }
});

module.exports = router;

Почему сообщение одинаковое и для «нет такого email», и для «неверный пароль»? Если бы они различались, по ответу можно было бы узнать, какие email зарегистрированы.

В исходном материале у маршрута входа нет try/catch. В Express 4 ошибка внутри async-обработчика (например, упала база) не попадёт в обработчик ошибок, и запрос просто повиснет. Express 5 такие ошибки ловит сам, но try/catch работает в обеих версиях.

Что делает jwt.sign

  1. Кодирует header {"alg":"HS256","typ":"JWT"} в base64url.
  2. Кодирует payload { id, iat, exp } в base64url. iat и exp библиотека добавит сама из expiresIn.
  3. Склеивает их через точку.
  4. Вычисляет подпись алгоритмом HMAC-SHA256 от этой строки с ключом JWT_SECRET.
  5. Добавляет подпись третьей частью.

Если кто-то заменит id в payload на чужой, но не знает секрета, он не сможет пересчитать подпись. Мы проверили: такой токен jwt.verify отклоняет с ошибкой invalid signature.
