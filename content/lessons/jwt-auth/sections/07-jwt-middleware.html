Защищённые маршруты: middleware

Middleware в Express — функция, которая выполняется до обработчика маршрута. Она может пропустить запрос дальше (next()) или сразу ответить сама. Для авторизации это идеальное место: проверяем токен один раз и подключаем проверку к нужным маршрутам.

// middleware/auth.js
const jwt = require('jsonwebtoken');

module.exports = function authMiddleware(req, res, next) {
  // 1. Достаём токен из заголовка "Authorization: Bearer xxx.yyy.zzz"
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    return res.status(401).json({ message: 'Токен не предоставлен' });
  }
  const token = authHeader.split(' ')[1];

  try {
    // 2. Проверяем подпись и срок
    const decoded = jwt.verify(token, process.env.JWT_SECRET, { algorithms: ['HS256'] });
    // decoded: { id: '664f2a', iat: ..., exp: ... }

    // 3. Кладём id в req, чтобы обработчик маршрута мог им пользоваться
    req.userId = decoded.id;
    next();
  } catch (err) {
    // подпись не совпала или срок истёк
    return res.status(401).json({ message: 'Невалидный или просроченный токен' });
  }
};

Что делает jwt.verify

  1. Разбивает токен на три части.
  2. Сам заново считает подпись от первых двух частей своим секретом.
  3. Сравнивает её с подписью в токене. Не совпала — ошибка invalid signature.
  4. Смотрит на exp. Срок прошёл — ошибка jwt expired.
  5. Если всё в порядке, возвращает payload как обычный объект.

Важная деталь: при неудаче jwt.verify выбрасывает исключение, а не возвращает false. Поэтому нужен try/catch, а не if.

Опция algorithms: ['HS256'] говорит библиотеке принимать только наш алгоритм. Её нет в исходном материале, но её советует документация jsonwebtoken: так сервер не примет токен, подписанный другим способом.

Подключаем к маршруту

// routes/user.js
const router = require('express').Router();
const authMiddleware = require('../middleware/auth');
const User = require('../models/User');

router.get('/profile', authMiddleware, async (req, res) => {
  try {
    // req.userId положил authMiddleware
    const user = await User.findById(req.userId).select('-password');
    if (!user) return res.status(404).json({ message: 'Пользователь не найден' });
    res.json(user);
  } catch (err) {
    res.status(500).json({ message: 'Ошибка сервера' });
  }
});

module.exports = router;

.select('-password') убирает хеш пароля из ответа. Пользователя всё равно ищем в базе: токен говорит, кто пришёл, но актуальные данные профиля лежат в базе. Пользователя могли удалить уже после того, как токен был выдан, — для этого и проверка на 404.

Собираем приложение

// app.js
require('dotenv').config();
const express = require('express');
const mongoose = require('mongoose');

const app = express();
app.use(express.json()); // без этого req.body будет undefined
app.use('/api/auth', require('./routes/auth'));
app.use('/api/user', require('./routes/user'));

mongoose.connect(process.env.MONGO_URI)
  .then(() => app.listen(3000, () => console.log('Server on :3000')))
  .catch(console.error);

dotenv подключается первой строкой: переменные из .env должны появиться раньше, чем их прочитает остальной код.
