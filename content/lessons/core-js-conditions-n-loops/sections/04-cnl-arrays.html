Массивы и матрицы без методов

Массивы можно создавать литералом [] и заполнять по индексу arr[i] = x — это не методы, а обычное присваивание. Запрещены push, map, reverse и все остальные вызовы.

10. getBalanceIndex — точка равновесия

 [1, 2, 5, 3, 0] => 2    => 1 + 2 === 3 + 0 then balance element is 5 and its index = 2
 [2, 3, 9, 5] => 2       => 2 + 3 === 5 then balance element is 9 and its index = 2
 [1, 2, 3, 4, 5] => -1   => no balance element

Ищем индекс, где сумма слева равна сумме справа. Наивно — для каждого индекса пересчитывать обе суммы (O(n²)). Лучше один раз посчитать общую сумму, затем идти слева направо: сумма справа = total − left − arr[i], а left копится по ходу. Получается O(n).

Особенность тестов: эталонное решение в тестах ищет точку начиная с индекса 1 и для массивов короче трёх элементов сразу возвращает −1. Проверка на 40 прогонах показала, что решение, которое находит и индекс 0 (когда сумма всех элементов справа равна нулю), иногда падает на случайных данных теста. Поэтому начинаем с left = arr[0] и i = 1.

function getBalanceIndex(arr) {
  if (arr.length < 3) return -1;
  let total = 0;
  for (let i = 0; i < arr.length; i += 1) total += arr[i];
  let left = arr[0];
  for (let i = 1; i < arr.length; i += 1) {
    const right = total - left - arr[i];
    if (left === right) return i;
    left += arr[i];
  }
  return -1;
}

11. getSpiralMatrix — спиральная матрица

       [
         [1, 2, 3],
 3  =>   [8, 9, 4],
         [7, 6, 5]
       ]
       [
         [1,  2,  3,  4],
 4  =>   [12, 13, 14, 5],
         [11, 16, 15, 6],
         [10, 9,  8,  7]
       ]

Держим четыре границы: top, bottom, left, right. Проходим верхнюю строку слева направо и сдвигаем top вниз, правый столбец сверху вниз и сдвигаем right влево, нижнюю строку справа налево, левый столбец снизу вверх. Повторяем, пока границы не сошлись. Дополнительные условия top <= bottom и left <= right во внутренних циклах не дают пройти центральную строку или столбец дважды. Подробнее о матрицах — урок «Алгоритмы и структуры данных».

function getSpiralMatrix(size) {
  const matrix = [];
  for (let i = 0; i < size; i += 1) matrix[i] = [];
  let top = 0;
  let bottom = size - 1;
  let left = 0;
  let right = size - 1;
  let value = 1;
  while (top <= bottom && left <= right) {
    for (let col = left; col <= right; col += 1) {
      matrix[top][col] = value;
      value += 1;
    }
    top += 1;
    for (let row = top; row <= bottom; row += 1) {
      matrix[row][right] = value;
      value += 1;
    }
    right -= 1;
    for (let col = right; col >= left && top <= bottom; col -= 1) {
      matrix[bottom][col] = value;
      value += 1;
    }
    bottom -= 1;
    for (let row = bottom; row >= top && left <= right; row -= 1) {
      matrix[row][left] = value;
      value += 1;
    }
    left += 1;
  }
  return matrix;
}

12. rotateMatrix — поворот на 90° на месте

 [                 [
   [1, 2, 3],        [7, 4, 1],
   [4, 5, 6],  =>    [8, 5, 2],
   [7, 8, 9]         [9, 6, 3]
 ]                 ]

Классический приём в два шага. Транспонирование: строки становятся столбцами (меняем m[i][j] и m[j][i] над диагональю). Затем каждую строку разворачиваем двумя указателями. Вместе это поворот по часовой стрелке без дополнительной матрицы — «на месте», как требует условие. Параметр функции сразу присваивается переменной m: правило airbnb no-param-reassign запрещает менять свойства параметров напрямую, а через локальную переменную линтер это допускает.

function rotateMatrix(matrix) {
  const m = matrix;
  const n = m.length;
  for (let i = 0; i < n; i += 1) {
    for (let j = i + 1; j < n; j += 1) {
      const tmp = m[i][j];
      m[i][j] = m[j][i];
      m[j][i] = tmp;
    }
  }
  for (let i = 0; i < n; i += 1) {
    for (let left = 0, right = n - 1; left < right; left += 1, right -= 1) {
      const tmp = m[i][left];
      m[i][left] = m[i][right];
      m[i][right] = tmp;
    }
  }
  return m;
}

13. sortByAsc — своя сортировка на месте

 [2, 9, 5]       => [2, 5, 9]
 [2, 9, 5, 9]    => [2, 5, 9, 9]
 [-2, 9, 5, -3]  => [-3, -2, 5, 9]

Тест скорости сортирует 7500 чисел от −100 до 100 и даёт на это 35 мс. Пузырьковая сортировка (O(n²), около 28 миллионов сравнений) не укладывается: на машине, где готовился урок, она заняла 41 мс, а решение ниже — около 2 мс. Нужна сортировка за O(n log n). Здесь — быстрая сортировка (quick sort) с трёхсторонним разбиением: элементы делятся на «меньше опорного», «равны» и «больше». Чисел всего 201 вариант, повторов очень много, и обычный quick sort на них деградирует, а трёхстороннее разбиение сразу откладывает все равные. Рекурсию заменяет свой стек границ отрезков (обычный массив, заполняемый по индексу), чтобы не упереться в глубину вызовов. Опорный элемент — середина отрезка.

function sortByAsc(arr) {
  const a = arr;
  const stack = [0, a.length - 1];
  let top = 2;
  while (top > 0) {
    top -= 2;
    const lo = stack[top];
    const hi = stack[top + 1];
    if (lo < hi) {
      const pivot = a[lo + Math.floor((hi - lo) / 2)];
      let lt = lo;
      let gt = hi;
      let i = lo;
      while (i <= gt) {
        if (a[i] < pivot) {
          const tmp = a[lt];
          a[lt] = a[i];
          a[i] = tmp;
          lt += 1;
          i += 1;
        } else if (a[i] > pivot) {
          const tmp = a[gt];
          a[gt] = a[i];
          a[i] = tmp;
          gt -= 1;
        } else {
          i += 1;
        }
      }
      stack[top] = lo;
      stack[top + 1] = lt - 1;
      stack[top + 2] = gt + 1;
      stack[top + 3] = hi;
      top += 4;
    }
  }
  return a;
}

14. shuffleChar — перемешивание символов

 '012345', 1 => '024135'
 'qwerty', 1 => 'qetwry'
 '012345', 2 => '024135' => '043215'
 'qwerty', 2 => 'qetwry' => 'qtrewy'
 '012345', 3 => '024135' => '043215' => '031425'
 'qwerty', 3 => 'qetwry' => 'qtrewy' => 'qrwtey'

Одна итерация переносит символы с нечётными индексами в конец: «012345» → «024135». Тест: строка из 1000 символов и 10 000 итераций за 30 мс. Честно повторять перестановку 10 000 раз — 10 миллионов операций со строками, слишком медленно.

Ключевое наблюдение: перестановка всегда одна и та же, она лишь переставляет позиции. Для каждой позиции k результата известно, откуда пришёл символ: из 2k для первой половины и из 2(k − half) + 1 для второй. Позиции распадаются на циклы — замкнутые цепочки, по которым символы ходят по кругу. Если цикл длины L, то после L итераций символы на нём возвращаются на место, поэтому достаточно сдвинуться на iterations % L шагов. Каждую позицию обрабатываем один раз — O(n) независимо от числа итераций: 1000 символов и 10 000 итераций заняли 0,2 мс.

function shuffleChar(str, iterations) {
  const n = str.length;
  const half = Math.ceil(n / 2);
  const from = [];
  for (let k = 0; k < n; k += 1) {
    from[k] = k < half ? 2 * k : 2 * (k - half) + 1;
  }
  const result = [];
  const visited = [];
  for (let start = 0; start < n; start += 1) {
    if (!visited[start]) {
      const cycle = [];
      let len = 0;
      let k = start;
      while (!visited[k]) {
        visited[k] = true;
        cycle[len] = k;
        len += 1;
        k = from[k];
      }
      const steps = iterations % len;
      for (let p = 0; p < len; p += 1) {
        result[cycle[p]] = str[cycle[(p + steps) % len]];
      }
    }
  }
  let out = '';
  for (let k = 0; k < n; k += 1) out += result[k];
  return out;
}

15. getNearestBigger — ближайшее большее из тех же цифр

12345    => 12354
123450   => 123504
12344    => 12434
123440   => 124034
1203450  => 1203504
90822    => 92028
321321   => 322113

Это классический алгоритм «следующая перестановка». Цифры раскладываем в массив арифметикой (младшая цифра — в digits[0]). Затем: 1) с конца числа (то есть с младших разрядов) ищем первую позицию i, где цифра меньше соседней справа — её можно увеличить; если такой нет, цифры уже убывают и большего числа не существует. 2) Среди цифр правее находим наименьшую, которая больше digits[i], и меняем их местами. 3) Хвост правее i разворачиваем, чтобы он стал наименьшим. Пример: 12344 → 12434. Тест скорости — 10 мс на 16-значном числе; алгоритм линейный по числу цифр.

function getNearestBigger(number) {
  const digits = [];
  let len = 0;
  let rest = number;
  while (rest > 0) {
    digits[len] = rest % 10;
    rest = (rest - digits[len]) / 10;
    len += 1;
  }
  let i = 1;
  while (i < len && digits[i] >= digits[i - 1]) i += 1;
  if (i === len) return number;
  let j = 0;
  while (digits[j] <= digits[i]) j += 1;
  let tmp = digits[i];
  digits[i] = digits[j];
  digits[j] = tmp;
  for (let left = 0, right = i - 1; left < right; left += 1, right -= 1) {
    tmp = digits[left];
    digits[left] = digits[right];
    digits[right] = tmp;
  }
  let result = 0;
  for (let k = len - 1; k >= 0; k -= 1) result = result * 10 + digits[k];
  return result;
}
