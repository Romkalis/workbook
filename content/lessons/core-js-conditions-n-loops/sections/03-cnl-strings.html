Строки и числа без методов

В этих задачах запрещены методы строк и массивов: тест ищет в коде .split(, .indexOf(, .includes( и десятки других. Остаются циклы, обращение к символу по индексу str[i], длина .length (это свойство, а не метод) и склейка строк через +.

5. convertToRomanNumerals — римские цифры до 39

 1   => I
 2   => II
 5   => V
 10  => X
 26  => XXVI

До 39 нужны только X (10), V (5) и I (1) плюс особые записи IX (9) и IV (4). Сначала выписываем десятки, вычитая по 10, затем остаток 0–9: 9 — это IX, иначе при остатке ≥ 5 ставим V, 4 — это IV, остальное добиваем единицами I.

function convertToRomanNumerals(num) {
  let result = '';
  let rest = num;
  while (rest >= 10) {
    result += 'X';
    rest -= 10;
  }
  if (rest === 9) return `${result}IX`;
  if (rest >= 5) {
    result += 'V';
    rest -= 5;
  }
  if (rest === 4) return `${result}IV`;
  while (rest > 0) {
    result += 'I';
    rest -= 1;
  }
  return result;
}

6. convertNumberToString — цифры словами

 '1'       => 'one'
 '10'      => 'one zero'
 '-10'     => 'minus one zero'
 '10.5'    => 'one zero point five'
 '10,5'    => 'one zero point five'
 '1950.2'  => 'one nine five zero point two'

Идём по символам и для каждого выбираем слово через switch — тест «optimal implementation» проверяет, что в коде есть switch. Точка и запятая обе превращаются в «point»: два case подряд без break ведут в одну ветку. Пробел ставим перед каждым словом, кроме первого, — без join и trim.

function convertNumberToString(numberStr) {
  let result = '';
  for (let i = 0; i < numberStr.length; i += 1) {
    let word;
    switch (numberStr[i]) {
      case '-':
        word = 'minus';
        break;
      case '.':
      case ',':
        word = 'point';
        break;
      case '0':
        word = 'zero';
        break;
      case '1':
        word = 'one';
        break;
      case '2':
        word = 'two';
        break;
      case '3':
        word = 'three';
        break;
      case '4':
        word = 'four';
        break;
      case '5':
        word = 'five';
        break;
      case '6':
        word = 'six';
        break;
      case '7':
        word = 'seven';
        break;
      case '8':
        word = 'eight';
        break;
      default:
        word = 'nine';
    }
    result += i === 0 ? word : ` ${word}`;
  }
  return result;
}

7. isPalindrome — палиндром ли строка

 'abcba'     => true
 '0123210'   => true
 'qweqwe'    => false

Два указателя: i идёт с начала, j с конца, пока не встретятся. При первом несовпадении — false. В for можно объявить и менять две переменные через запятую. Это быстрее, чем перевернуть строку целиком, и не требует методов.

function isPalindrome(str) {
  for (let i = 0, j = str.length - 1; i < j; i += 1, j -= 1) {
    if (str[i] !== str[j]) return false;
  }
  return true;
}

8. getIndexOf — индекс символа

 'qwerty', 'q'     => 0
 'qwerty', 't'     => 4
 'qwerty', 'Q'     => -1
 'qwerty', 'p'     => -1

Своя реализация indexOf: перебираем символы и возвращаем первый совпавший индекс, если не нашли — −1. Сравнение учитывает регистр: «Q» в «qwerty» не найдётся.

function getIndexOf(str, letter) {
  for (let i = 0; i < str.length; i += 1) {
    if (str[i] === letter) return i;
  }
  return -1;
}

9. isContainNumber — есть ли цифра в числе

 123450, 5   => true
 123450, 1   => true
 123450, 0   => true
 12345, 0    => false
 12345, 6    => false

Без перевода в строку цифры достаём арифметикой: n % 10 — последняя цифра, а (n - n % 10) / 10 отбрасывает её (то же, что целочисленное деление; Math.floor здесь не нужен). Цикл do...while выполняется хотя бы раз — это важно для числа 0, у которого единственная цифра 0.

function isContainNumber(num, digit) {
  let rest = num < 0 ? -num : num;
  do {
    if (rest % 10 === digit) return true;
    rest = (rest - (rest % 10)) / 10;
  } while (rest > 0);
  return false;
}
